<script>
// Constants for Logistic Growth Function Simulation
const L = 250; // Carrying capacity / Maximum potential weight in grams
const k = 0.22; // Growth coefficient rate
const x0 = 15; // Midpoint day of maximum accelerated growth

// Generate Dataset Array for Day 1 through Day 30
const days = Array.from({length: 30}, (_, i) => i + 1);

// Standard Logistic Equation f(x)
function calculateBiomass(x) {
return L / (1 + Math.exp(-k * (x - x0)));
}

// Analytical First Derivative f'(x)
function calculateGrowthRate(x) {
const fx = calculateBiomass(x);
return k * fx * (1 - (fx / L));
}

// Map calculated values into separate operational arrays
const biomassData = days.map(day => calculateBiomass(day));
const velocityData = days.map(day => calculateGrowthRate(day));

// Initialize and Design Chart.js instance
const ctx = document.getElementById('growthChart').getContext('2d');

// Gradient styling structures
const bioGradient = ctx.createLinearGradient(0, 0, 0, 400);
bioGradient.addColorStop(0, 'rgba(16, 185, 129, 0.4)');
bioGradient.addColorStop(1, 'rgba(16, 185, 129, 0)');

const growthChart = new Chart(ctx, {
type: 'line',
data: {
labels: days.map(d => `Day ${d}`),
datasets: [
{
label: 'Cumulative Biomass f(x) (grams)',
data: biomassData,
borderColor: '#10b981',
borderWidth: 3,
pointBackgroundColor: '#10b981',
pointHoverRadius: 8,
pointHoverBackgroundColor: '#ffffff',
fill: true,
backgroundColor: bioGradient,
tension: 0.2,
yAxisID: 'y'
},
{
label: "Growth Velocity f'(x) (g/day)",
data: velocityData,
borderColor: '#2dd4bf',
borderWidth: 2,
borderDash: [5, 5],
pointBackgroundColor: '#2dd4bf',
pointHoverRadius: 8,
pointHoverBackgroundColor: '#ffffff',
fill: false,
tension: 0.2,
yAxisID: 'y1'
}
]
},
options: {
responsive: true,
maintainAspectRatio: false,
plugins: {
legend: { display: false } // Custom legend built into HTML markup
},
scales: {
x: {
grid: { color: 'rgba(255, 255, 255, 0.05)' },
ticks: { color: '#94a3b8' }
},
y: {
position: 'left',
title: { display: true, text: 'Total Weight (grams)', color: '#10b981' },
grid: { color: 'rgba(255, 255, 255, 0.05)' },
ticks: { color: '#94a3b8' }
},
y1: {
position: 'right',
title: { display: true, text: 'Velocity (g/day)', color: '#2dd4bf' },
grid: { drawOnChartArea: false },
ticks: { color: '#94a3b8' }
}
},
// Trigger customized interactive state upon data node click mechanics
onClick: (e, elements) => {
if (elements.length > 0) {
const index = elements[0].index;
const daySelected = days[index];
const weight = biomassData[index];
const velocity = velocityData[index];

renderTelemetry(daySelected, weight, velocity);
}
},
// Smooth introductory presentation animation configuration
animation: {
duration: 2000,
easing: 'easeOutQuart'
}
}
});

// Telemetry Interaction Engine Output Function
function renderTelemetry(day, weight, velocity) {
document.getElementById('promptState').classList.add('hidden');
document.getElementById('dataState').classList.remove('hidden');

document.getElementById('telemetryDay').innerText = `Day ${day}`;
document.getElementById('telemetryWeight').innerText = `${weight.toFixed(1)}g`;
document.getElementById('telemetryRate').innerText = `+${velocity.toFixed(2)} g/day`;

// Dynamic algorithmic interpretation engine dependent on mathematical properties of derivative
let narrative = "";
if (day < 10) {
narrative = "The crop is in its early lag phase. The low derivative values show slow metabolic growth. Fertilizer requirements are minimal right now.";
} else if (day >= 10 && day <= 18) {
narrative = "Peak derivative acceleration! The crop is processing nutrients at maximum cellular velocity. Crucial window to maintain optimal LED light levels.";
} else {
narrative = "The growth velocity curve is flattening out (derivative approaches 0). The plant is mature and ready for harvest processing.";
}

document.getElementById('telemetryInsight').innerText = narrative;
}
